participant "Audio Player" as Audio

OS -> Serv: OnLocationChanged(Location loc)
Serv -> Calc: CheckGeofences(loc.Latitude, loc.Longitude)

    Calc -> UI: HighlightPoiOnMap(poi.Id)
    Calc -> Audio: CallCrossTTS(poi.Description, lang)
    Audio --> OS: PlayAudioStream()
else distance > poi.Radius + 30m
    Calc -> Calc: ResetAutoPlayFlag(poi.Id) = false
end

DB --> Web: DocumentSnapshot<PoiModel>
